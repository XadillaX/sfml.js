RenderWindow · CircleShape · Color

Your first SFML window.

Create a RenderWindow, draw a CircleShape, then display the frame. SFML handles the window and rendering; your JavaScript drives the loop. TypeScript declarations are included.

Follow the quick start

Run this in a graphical desktop session. Escape closes the window; setImmediate leaves room for Node.js I/O between frames.

JS hello.cjsnode hello.cjs
const sf = require('sfml.js');
const win = new sf.RenderWindow(
  new sf.VideoMode(new sf.Vector2u(800, 600)),
  'Hello, sfml.js',
);
const circle = new sf.CircleShape({ radius: 80 });
circle.setPosition(new sf.Vector2f(320, 220));
circle.setFillColor(new sf.Color(108, 190, 82));
win.setFramerateLimit(60);

function frame() {
  win.handleEvents({
    Closed: () => win.close(),
    KeyPressed: e => {
      if (e.code === sf.Keyboard.Key.Escape) win.close();
    },
  });
  if (!win.isOpen()) return;
  win.clear(new sf.Color(24, 28, 32));
  win.draw(circle);
  win.display();
  setImmediate(frame);
}
frame();
