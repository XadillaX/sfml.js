RenderWindow · CircleShape · Color

첫 SFML 창 만들기.

RenderWindow를 만들고 CircleShape를 그린 다음 프레임을 표시합니다. SFML은 창과 렌더링을, JavaScript는 루프를 담당합니다. TypeScript 타입 선언도 포함되어 있습니다.

빠른 시작 따라 하기

데스크톱 그래픽 환경에서 실행하세요. Esc로 창을 닫습니다. setImmediate는 프레임 사이에 Node.js I/O를 처리할 기회를 줍니다.

JS hello.cjsnode hello.cjs
const sf = require('sfml.js');
const win = new sf.RenderWindow(
  new sf.VideoMode(
    new sf.Vector2u(800, 600),
  ),
  'Hello, sfml.js',
);
const circle = new sf.CircleShape({ radius: 80 });
circle.setPosition(new sf.Vector2f(320, 220));
circle.setFillColor(new sf.Color(108, 190, 82));
win.setFramerateLimit(60);

function frame() {
  win.handleEvents({
    Closed: () => win.close(),
    KeyPressed: e => {
      if (e.code === sf.Keyboard.Key.Escape) {
        win.close();
      }
    },
  });
  if (!win.isOpen()) {
    return;
  }
  win.clear(new sf.Color(24, 28, 32));
  win.draw(circle);
  win.display();
  setImmediate(frame);
}
frame();
